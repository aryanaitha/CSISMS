package com.csims.config;

@Configuration @EnableWebSecurity @EnableMethodSecurity
public class SecurityConfig implements WebMvcConfigurer {
    @Bean public PasswordEncoder passwordEncoder() { return new BCryptPasswordEncoder(); }

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        return http.cors(c -> c.configurationSource(corsConfigurationSource()))
            .csrf(c -> c.disable())
            .exceptionHandling(e -> e.authenticationEntryPoint((req, res, ex) -> {
                res.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
                res.setContentType(MediaType.APPLICATION_JSON_VALUE);
                res.getWriter().write("{"success":false,"message":"Authentication required"}");
            }))
            .sessionManagement(s -> s.sessionFixation().changeSessionId().maximumSessions(1))
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/", "/index.html", "/api/auth/login",
                    "/api/auth/register-request", "/api/auth/forgot-password",
                    "/api/health", "/swagger-ui.html").permitAll()
                .requestMatchers("/api/*").authenticated().anyRequest().permitAll())
            .build();
    }
}

public class AppExceptions {
    public static class ApiException extends RuntimeException {
        private final HttpStatus status;
        public ApiException(String m, HttpStatus s) { super(m); this.status = s; }
        public HttpStatus getStatus() { return status; }
    }
    public static class NotFoundException extends ApiException {

    }

@RestControllerAdvice
public class GlobalExceptionHandler {
    @ExceptionHandler(AppExceptions.ApiException.class)
    public ResponseEntity<ApiResponse<Void>> handleApiException(AppExceptions.ApiException e) {
        return ResponseEntity.status(e.getStatus()).body(ApiResponse.error(e.getMessage()));
    }
    @ExceptionHandler(AccessDeniedException.class)
    public ResponseEntity<ApiResponse<Void>> handleDenied(AccessDeniedException e) {
        return ResponseEntity.status(HttpStatus.FORBIDDEN).body(ApiResponse.error(
            "Access denied: You do not have permission to perform this action."));
    }
    @ExceptionHandler({BadCredentialsException.class, LockedException.class})
    public ResponseEntity<ApiResponse<Void>> handleAuthFailures(Exception e) {
        return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
            .body(ApiResponse.error(e.getMessage()));
    }
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiResponse<Void>> handleValidation(MethodArgumentNotValidException e) {
        String msg = e.getBindingResult().getFieldErrors().stream()
            .map(FieldError::getDefaultMessage).collect(Collectors.joining("; "));
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(ApiResponse.error(msg));
    }
}

6. Utility Classes
package com.csims.util;

public final class Utils {
    private Utils() {}

    // RFC-4180 Compliant CSV Generation with UTF-8 byte stream output
    public static byte[] generateCsv(List<String> headers, List<List<String>> rows)
            throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        try (PrintWriter w = new PrintWriter(
                new OutputStreamWriter(baos, StandardCharsets.UTF_8))) {
            if (headers != null && !headers.isEmpty()) w.println(toCsvRow(headers));
            if (rows != null) { for (List<String> r : rows) w.println(toCsvRow(r)); }
            w.flush();
        }
        return baos.toByteArray();
    }

    private static String toCsvRow(List<String> cols) {
        return cols.stream().map(Utils::escapeCsv).collect(Collectors.joining(","));
    }

    private static String escapeCsv(String v) {
        if (v == null) return """";
        return """ + v.replace(""", """") + """;
    }
}

package com.csims.repository;

@Repository
public class UserRepository {
    private final JdbcTemplate jdbcTemplate;
    public UserRepository(JdbcTemplate jdbcTemplate) { this.jdbcTemplate = jdbcTemplate; }

    // Parameterized RowMapper pattern (strictly JdbcTemplate, no JPA/Hibernate)
    private final RowMapper<User> userRowMapper = (rs, rowNum) -> {
        User u = new User();
        u.setId(rs.getLong("user_id")); u.setUsername(rs.getString("username"));
        u.setPasswordHash(rs.getString("password_hash")); u.setFullName(rs.getString("full_name"));
        u.setRoleId(rs.getInt("role_id")); u.setActive(rs.getBoolean("is_active"));
        u.setFailedLoginAttempts(rs.getInt("failed_attempts"));
        return u;
    };

    public Optional<User> findByUsernameOrEmail(String val) {
        String sql = "SELECT u.*, r.role_name FROM users u JOIN roles r ON u.role_id = r.role_id "
                   + "WHERE u.username = ? OR u.email = ?";
        return jdbcTemplate.query(sql, userRowMapper, val, val).stream().findFirst();
    }
}

8. Authentication Service
package com.csims.service;

@Service
public class AuthService {
    private final UserRepository userRepository;
    private final SystemService systemService;
    private final PasswordEncoder passwordEncoder;

    public UserDto login(LoginRequest req, HttpServletRequest request) {
        User user = userRepository.findByUsernameOrEmail(req.getUsername())
            .orElseThrow(() -> new BadCredentialsException("Invalid username or password"));
        if (!user.isActive()) throw new DisabledException("Account deactivated.");
        if (user.getLockedUntil() != null && LocalDateTime.now().isBefore(user.getLockedUntil()))
            throw new LockedException("Account locked. Try again later.");
        if (!passwordEncoder.matches(req.getPassword(), user.getPasswordHash())) {
            userRepository.incrementFailedLogins(user.getId());
            if (user.getFailedLoginAttempts() + 1 >= 5) {
                userRepository.lockAccount(user.getId(), LocalDateTime.now().plusMinutes(15));
                throw new LockedException("Account locked for 15 min due to 5 failed attempts.");
            }
            throw new BadCredentialsException("Invalid username or password");
        }
        userRepository.resetFailedLoginsAndLock(user.getId());
        userRepository.updateLastLogin(user.getId());
        Set<String> perms = userRepository.findPermissionNamesByRoleId(user.getRoleId());
        SecurityContextHolder.getContext().setAuthentication(
            new UsernamePasswordAuthenticationToken(new CustomUserDetails(user, perms), null,
                AuthorityUtils.createAuthorityList(perms.toArray(new String[0]))));
        request.getSession(true).setAttribute(
            HttpSessionSecurityContextRepository.SPRING_SECURITY_CONTEXT_KEY,
            SecurityContextHolder.getContext());
        systemService.log(user.getId(), user.getUsername(), user.getRoleName(), "LOGIN_SUCCESS",
            "AUTH", "USER", String.valueOf(user.getId()), null, null, "SUCCESS", "Logged in");
        return toDto(user, perms);
    }

    public void logout(HttpServletRequest req) {
        SecurityContextHolder.clearContext();
        HttpSession session = req.getSession(false);
        if (session != null) session.invalidate();
    }
}

package com.csims.service;

@Service
public class AdminService {
    private final UserRepository userRepository;
    private final PasswordEncoder passwordEncoder;
    private final SystemService systemService;

    @Transactional
    public Long createUser(User user, String rawPwd, Long operatorId, String operatorUsername) {
        if (userRepository.existsByUsername(user.getUsername()))
            throw new ApiException("Username already exists", HttpStatus.CONFLICT);
        user.setPasswordHash(passwordEncoder.encode(rawPwd));
        user.setMustChangePassword(true); user.setActive(true);
        Long id = userRepository.createUser(user);
        systemService.log(operatorId, operatorUsername, null, "USER_CREATE", "USER", "USER",
            String.valueOf(id), null, user.getUsername(), "SUCCESS", "Created user");
        return id;
    }

    @Transactional
    public void toggleUserStatus(Long id, boolean active, Long opId, String opUsername) {
        if (id == 1 && !active)
            throw new ApiException("Cannot deactivate primary Admin", HttpStatus.BAD_REQUEST);
        User u = getUserById(id); u.setActive(active); userRepository.updateUser(u);
    }

    @Transactional
    public void approveRequest(Long id, Long reviewerId, String reviewerUser, String reviewNotes) {
        AccountRequest req = getRequestById(id);
        if (!"PENDING".equalsIgnoreCase(req.getStatus()))
            throw new ApiException("Request already processed", HttpStatus.BAD_REQUEST);
        if ("REGISTRATION".equalsIgnoreCase(req.getRequestType())) {
            User u = new User(); u.setUsername(req.getUsername()); u.setEmail(req.getEmail());
            u.setFullName(req.getFullName()); u.setRoleId(req.getRequestedRoleId());
            u.setPasswordHash(passwordEncoder.encode("Welcome@123"));
            u.setActive(true); u.setMustChangePassword(true);
            Long userId = userRepository.createUser(u);
            systemService.notify(userId, "Approved", "Temp pwd: Welcome@123", "SUCCESS", null);
        }
        userRepository.updateAccountRequestReview(id, "APPROVED", reviewerId, reviewNotes);
    }
}

13. Inventory and Material Stock Service
package com.csims.service;

@Service
public class InventoryService {
    private final InventoryRepository inventoryRepository;
    private final SystemService systemService;

    @Transactional
    public void recordReceipt(Integer matId, BigDecimal qty, String ref, String remarks,
            Long opId, String opUser) {
        inventoryRepository.adjustStock(matId, qty);
        InventoryTransaction tx = new InventoryTransaction();
        tx.setMaterialId(matId); tx.setTransactionType("RECEIPT"); tx.setQuantity(qty);
        tx.setReferenceNo(ref); tx.setRemarks(remarks); tx.setPerformedBy(opId);
        inventoryRepository.recordTransaction(tx);
        systemService.log(opId, opUser, null, "INVENTORY_RECEIPT", "INVENTORY", "MATERIAL",
            String.valueOf(matId), null, String.valueOf(qty), "SUCCESS", "Stock received");
    }

    @Transactional
    public void recordIssue(Integer matId, BigDecimal qty, Integer projId, String ref,
            String remarks, Long opId, String opUser) {
        Material m = getMaterialById(matId);
        // Atomic check: issue fails if stock < quantity -> rollback
        if (m.getQuantityOnHand().compareTo(qty) < 0) {
            throw new ApiException("Insufficient stock for " + m.getName() + ". Available: "
                + m.getQuantityOnHand() + ", Requested: " + qty, HttpStatus.BAD_REQUEST);
        }
        inventoryRepository.adjustStock(matId, qty.negate());
        InventoryTransaction tx = new InventoryTransaction();
        tx.setMaterialId(matId); tx.setTransactionType("ISSUE"); tx.setQuantity(qty);
        tx.setProjectId(projId); tx.setReferenceNo(ref); tx.setPerformedBy(opId);
        inventoryRepository.recordTransaction(tx);

        if (m.getQuantityOnHand().subtract(qty).compareTo(m.getMinStock()) <= 0) {
            systemService.notify(1L, "Low Stock Warning", "Material " + m.getName()
                + " reached threshold.", "WARNING", "/inventory");
        }
    }
}

package com.csims.service;

@Service
public class SafetyService {
    private final SafetyRepository safetyRepository;
    private final SystemService systemService;

    @Transactional
    public Integer conductInspection(SafetyInspection si, Long inspId, String inspUser) {
        si.setInspectorId(inspId);
        double total = si.getPpeScore() + si.getWorkerSafetyScore() + si.getEquipmentScore()
                     + si.getWorkAreaScore() + si.getEmergencyScore();
        BigDecimal overall = BigDecimal.valueOf(total / 5.0).setScale(2, RoundingMode.HALF_UP);
        si.setOverallScore(overall);
        Integer id = safetyRepository.createInspection(si);
        if (overall.compareTo(BigDecimal.valueOf(70)) < 0) {
            systemService.notify(inspId, "Safety Score Alert (< 70%)",
                "Inspection ID " + id + " scored " + overall + "%. Urgent action required.",
                "DANGER", "/safety");
        }
        return id;
    }

    @Transactional
    public Integer reportIncident(Incident inc, Long reporterId, String reporterUsername) {
        inc.setReportedBy(reporterId);
        Integer id = safetyRepository.createIncident(inc);
        if ("FATALITY".equalsIgnoreCase(inc.getSeverity())
                || "INJURY".equalsIgnoreCase(inc.getSeverity())) {
            systemService.notify(reporterId, "EMERGENCY: " + inc.getSeverity() + " Reported!",
                "Accident at project " + inc.getProjectId() + ": " + inc.getTitle(), "DANGER",
                "/incidents");
        }
        return id;
    }
}

@RequestMapping("/api/projects")
public class ProjectController {
    private final ProjectService projectService;

    @PostMapping
    @PreAuthorize("hasAuthority('PROJECT_CREATE') or hasRole('Administrator')")
    public ResponseEntity<ApiResponse<Integer>> createProject(@Valid @RequestBody Project p,
            @AuthenticationPrincipal CustomUserDetails user) {
        Integer id = projectService.createProject(p, user.getId(), user.getUsername());
        return ResponseEntity.ok(ApiResponse.success("Project created successfully", id));
    }
}
